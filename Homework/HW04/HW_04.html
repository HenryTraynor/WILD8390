set.seed(314159) # for reproducibility

nSites <- 100
nVisits <- 4

forest <- factor(sample(c("Hardwood", "Mixed", "Pine"), nSites, replace=TRUE))
forestMixed <- ifelse(forest=="Mixed", 1, 0)        ## Dummy
forestPine <- ifelse(forest=="Pine", 1, 0)          ## Dummy
temp <- matrix(rnorm(nSites*nVisits), nrow=nSites, ncol=nVisits)

# true coeffs
beta0 <- 0
beta1 <- -1
beta2 <- 1
lambda <- exp(beta0 + beta1*forestMixed + beta2*forestPine)
alpha0 <- -2
alpha1 <- 1
p <- plogis(alpha0 + alpha1*temp)

# true lambda plots
lambda.true <- c(
  Hardwood = exp(beta0),
  Mixed = exp(beta0 + beta1),
  Pine = exp(beta0 + beta2)
)
lambda.true

## 1.0000000 0.3678794 2.7182818
# simulate occupancy and detection data
N <- rpois(nSites, lambda=lambda)         ## local abundance 
y <- matrix(NA, nrow=nSites, ncol=nVisits)
for(i in 1:nSites) {
    y[i,] <- rbinom(nVisits, size=N[i], prob=p[i,])
}

# summary statistics
maxCounts <- apply(y, 1, max)

# prep data in named list
jags.data <- list(y=y,
                  forestMixed=forestMixed,
                  forestPine=forestPine,
                  nSites=nSites, 
                  temp=temp,
                  nOccasions=nVisits)
# initial values for MCMC
jags.inits <- function() {
    list(beta0=rnorm(1), alpha0=rnorm(1), N=maxCounts)
}
# parameters to save
jags.pars <- c("beta0", "beta1", "beta2",
               "alpha0", "alpha1", "totalAbundance")

# run model
library(jagsUI)
jags.post.samples <- jags.basic(data=jags.data, 
                                inits=jags.inits,
                                parameters.to.save=c(jags.pars,"N"), 
                                model.file="Nmix-model-covs.jag",
                                n.chains=3,
                                n.adapt=100, 
                                n.burnin=0,
                                n.iter=6000,
                                parallel=TRUE)
